        <!-- 1. The visual display (representing State) -->
        <div id="status" class="status-box danger">
            STATUS: DANGER (0V) - Fence Offline!
        </div>

        <!-- 2. The Button (Triggering our event) -->
        <button id="securityButton" onclick="toggleSecurity()">Activate Fence (130V)</button>
    </div>

    <script>
        // 3. The State: Starts as 'false' (offline / dangerous)
        let isSecure = false;

        // 4. The Event Handler: Runs automatically when the button is clicked!
        function toggleSecurity() {
            // Flip the state to its opposite! (True becomes False, False becomes True)
            isSecure = !isSecure;

            // Find the visual elements on our webpage
            const statusDiv = document.getElementById("status");
            const button = document.getElementById("securityButton");

            // 5. The Condition: Check our state and update the screen!
            if (isSecure) {
                statusDiv.innerText = "STATUS: SECURE (130V) - Fence Active!";
                statusDiv.className = "status-box secure";
                button.innerText = "Deactivate Fence";
                button.style.backgroundColor = "#4CAF50";
            } else {
                statusDiv.innerText = "STATUS: DANGER (0V) - Fence Offline!";
                statusDiv.className = "status-box danger";
                button.innerText = "Activate Fence (130V)";
                button.style.backgroundColor = "#FF5252";
            }
        }
    </script>
